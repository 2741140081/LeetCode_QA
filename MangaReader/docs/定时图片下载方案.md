图片定时下载方案.md
高效定时图片下载保存解决方案（Java 17 + Spring Boot + MyBatis + MySQL 8.0）

方案概述
采用 Spring Boot 定时任务扫描数据库 + 多线程异步下载 + 连接池 + 断点续传 + 状态自动管理 的架构，实现完全由定时任务驱动的自动化、高可靠图片批量下载能力。无需手动触发启动，服务启动后自动按配置的频率扫描数据库中的待下载任务，通过多线程并行执行下载，内置失败重试机制，异常场景下自动恢复任务状态，支持断点续传避免重复下载，适配万级及以上规模的图片批量下载场景。

1. 数据库表设计
```sql
CREATE TABLE `image_download_info` (
    `id` BIGINT NOT NULL AUTO_INCREMENT,
    `image_url` VARCHAR(1024) NOT NULL COMMENT '图片下载地址',
    `local_path` VARCHAR(512) DEFAULT NULL COMMENT '本地保存路径',
    `status` TINYINT NOT NULL DEFAULT 0 COMMENT '状态: 0-待下载, 1-下载中, 2-成功, 3-失败',
    `file_size` BIGINT DEFAULT NULL COMMENT '文件大小(字节)',
    `downloaded_size` BIGINT DEFAULT 0 COMMENT '已下载大小(字节,用于断点续传)',
    `retry_count` INT NOT NULL DEFAULT 0 COMMENT '重试次数',
    `error_msg` VARCHAR(512) DEFAULT NULL COMMENT '错误信息',
    `last_trigger_time` DATETIME DEFAULT NULL COMMENT '最后定时任务触发时间',
    `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `update_time` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    INDEX `idx_status` (`status`),
    INDEX `idx_create_time` (`create_time`),
    INDEX `idx_retry_status` (`retry_count`, `status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='定时图片下载信息表';
```

```java
package com.example.imagedownload.mapper;

import com.example.imagedownload.entity.ImageDownloadInfo;
import org.apache.ibatis.annotations.*;

import java.util.List;

@Mapper
public interface ImageDownloadMapper {

    @Select("SELECT * FROM image_download_info WHERE status = {status} ORDER BY id ASC LIMIT {limit}")
    List<ImageDownloadInfo> selectPendingTasks(@Param("status") Integer status, @Param("limit") Integer limit);

    @Select("SELECT * FROM image_download_info WHERE status = 1 AND last_trigger_time < DATE_SUB(NOW(), INTERVAL {timeout} SECOND) LIMIT {limit}")
    List<ImageDownloadInfo> selectTimeoutDownloadingTasks(@Param("timeout") Integer timeout, @Param("limit") Integer limit);

    @Update("UPDATE image_download_info SET status = {status}, downloaded_size = {downloadedSize}, " +
            "error_msg = {errorMsg}, update_time = NOW() WHERE id = {id}")
    int updateStatus(@Param("id") Long id, @Param("status") Integer status,
                     @Param("downloadedSize") Long downloadedSize, @Param("errorMsg") String errorMsg);

    @Update("UPDATE image_download_info SET status = 1, last_trigger_time = NOW() WHERE id = {id}")
    int markAsDownloading(@Param("id") Long id);

    @Update("UPDATE image_download_info SET status = 2, local_path = {localPath}, " +
            "file_size = {fileSize}, update_time = NOW() WHERE id = {id}")
    int markAsSuccess(@Param("id") Long id, @Param("localPath") String localPath, @Param("fileSize") Long fileSize);

    @Select("SELECT COUNT(*) FROM image_download_info WHERE status = 0")
    int countPendingTasks();

    @Select("SELECT COUNT(*) FROM image_download_info WHERE status = 1")
    int countDownloadingTasks();

    @Update("UPDATE image_download_info SET retry_count = retry_count + 1 WHERE id = {id}")
    int incrementRetryCount(@Param("id") Long id);

    @Select("SELECT * FROM image_download_info WHERE id = {id}")
    ImageDownloadInfo selectById(@Param("id") Long id);

    @Insert("INSERT INTO image_download_info(image_url, status) VALUES({imageUrl}, 0)")
```

```java
package com.example.imagedownload.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.scheduling.annotation.EnableAsync;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;

import java.util.concurrent.Executor;
import java.util.concurrent.ThreadPoolExecutor;

@Configuration
@EnableAsync
public class AsyncConfig {

    private final DownloadConfig downloadConfig;

    public AsyncConfig(DownloadConfig downloadConfig) {
        this.downloadConfig = downloadConfig;
    }

    @Bean("downloadExecutor")
    public Executor downloadExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(downloadConfig.getCoreThreads());
        executor.setMaxPoolSize(downloadConfig.getMaxThreads());
        executor.setQueueCapacity(100);
        executor.setThreadNamePrefix("img-dl-");
        executor.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());
        executor.setWaitForTasksToCompleteOnShutdown(true);
        executor.setAwaitTerminationSeconds(60);
        executor.initialize();
        return executor;
    }
}
```

7. 定时任务核心实现
```java
// ImageDownloadScheduledTask.java
package com.example.imagedownload.task;

import com.example.imagedownload.config.DownloadConfig;
import com.example.imagedownload.entity.ImageDownloadInfo;
import com.example.imagedownload.mapper.ImageDownloadMapper;
import com.example.imagedownload.service.ImageDownloadService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.concurrent.Executor;

@Slf4j
@Component
@RequiredArgsConstructor
public class ImageDownloadScheduledTask {

    private final ImageDownloadMapper mapper;
    private final DownloadConfig config;
    private final Executor downloadExecutor;
    private final ImageDownloadService downloadService;

    /**
     * 定时扫描待下载任务, 默认每2秒执行一次
     */
    @Scheduled(cron = "${image.download.scan-cron:0/2 * * * * ?}")
    public void scanPendingTasks() {
        List<ImageDownloadInfo> tasks = mapper.selectPendingTasks(0, config.getBatchSize());
        if (tasks.isEmpty()) {
            return;
        }
        log.info("定时任务扫描到 {} 个待下载图片任务", tasks.size());

        for (ImageDownloadInfo task : tasks) {
            // 数据库乐观锁标记为下载中, 防止分布式环境下重复抢占任务
            int updated = mapper.markAsDownloading(task.getId());
            if (updated == 0) {
                continue;
            }
            // 提交至多线程池异步下载
            downloadExecutor.execute(() -> downloadService.downloadImage(task));
        }
    }

    /**
     * 定时扫描僵死超时任务, 默认每30秒执行一次
     * 处理服务重启、网络异常导致的长时间卡在下载中的任务, 自动重置为待下载状态
     */
    @Scheduled(fixedDelay = 30000)
    public void recoverTimeoutTasks() {
        List<ImageDownloadInfo> timeoutTasks = mapper.selectTimeoutDownloadingTasks(config.getDownloadTimeout(), 100);
        if (timeoutTasks.isEmpty()) {
            return;
        }
        log.warn("扫描到 {} 个超时僵死的下载任务, 自动重置重试", timeoutTasks.size());

        for (ImageDownloadInfo task : timeoutTasks) {
            downloadService.handleFailure(task.getId(), "下载任务超时,触发定时自动恢复");
        }
    }
}
```

ImageDownloadServiceImpl.java
```java
// ImageDownloadServiceImpl.java
package com.example.imagedownload.service.impl;

import com.example.imagedownload.config.DownloadConfig;
import com.example.imagedownload.entity.ImageDownloadInfo;
import com.example.imagedownload.mapper.ImageDownloadMapper;
import com.example.imagedownload.service.ImageDownloadService;
import lombok.extern.slf4j.Slf4j;
import okhttp3.*;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.stereotype.Service;

import java.io.*;
import java.net.URL;
import java.nio.file.*;
import java.util.*;
import java.util.concurrent.Executor;
import java.util.concurrent.atomic.AtomicBoolean;

@Slf4j
@Service
public class ImageDownloadServiceImpl implements ImageDownloadService {

    private final ImageDownloadMapper mapper;
    private final DownloadConfig config;
    private final OkHttpClient httpClient;
    private final Path saveBasePath;
    private final Executor downloadExecutor;

    private final AtomicBoolean running = new AtomicBoolean(false);

    public ImageDownloadServiceImpl(ImageDownloadMapper mapper,
                                     DownloadConfig config,
                                     @Qualifier("okHttpClient") OkHttpClient httpClient,
                                     @Qualifier("downloadExecutor") Executor downloadExecutor) {
        this.mapper = mapper;
        this.config = config;
        this.httpClient = httpClient;
        this.downloadExecutor = downloadExecutor;
        this.saveBasePath = Paths.get(config.getSaveBasePath());
        // 启动时自动创建保存目录
        try {
            Files.createDirectories(saveBasePath);
        } catch (IOException e) {
            throw new RuntimeException("创建图片保存目录失败: " + config.getSaveBasePath(), e);
        }
    }

    /**
     * 核心下载逻辑 - 支持断点续传
     */
    @Override
    public void downloadImage(ImageDownloadInfo task) {
        long taskId = task.getId();
        log.info("[Task-{}] 开始执行图片定时下载: {}", taskId, task.getImageUrl());

        Request.Builder requestBuilder = new Request.Builder()
                .url(task.getImageUrl())
                .header("User-Agent", "Mozilla/5.0 ImageScheduledDownloader/1.0");

        // 如果有未完成的已下载数据, 自动启用断点续传
        if (task.getDownloadedSize() != null && task.getDownloadedSize() > 0) {
            requestBuilder.header("Range", "bytes=" + task.getDownloadedSize() + "-");
        }

        Request request = requestBuilder.build();

        try (Response response = httpClient.newCall(request).execute()) {
            if (!response.isSuccessful()) {
                handleFailure(taskId, "HTTP 状态码异常: " + response.code());
                return;
            }

            long contentLength = response.body().contentLength();
            long startPos = task.getDownloadedSize() != null ? task.getDownloadedSize() : 0;

            // 生成合法文件名, 避免冲突
            String fileName = extractFileName(task.getImageUrl(), response, taskId);
            Path targetPath = saveBasePath.resolve(fileName);
            targetPath = resolveConflict(targetPath);

            // 使用NIO缓冲流追加写入, 适配断点续传场景
            try (BufferedOutputStream fos = new BufferedOutputStream(
                    new FileOutputStream(targetPath.toFile(), true), config.getBufferSize())) {

                long totalDownloaded = startPos;
                try (BufferedInputStream bis = new BufferedInputStream(
                        response.body().byteStream(), config.getBufferSize())) {

                    byte[] buffer = new byte[config.getBufferSize()];
                    int bytesRead;
                    while ((bytesRead = bis.read(buffer)) != -1) {
                        fos.write(buffer, 0, bytesRead);
                        totalDownloaded += bytesRead;

                        // 每累计下载1MB更新一次数据库进度, 大幅降低数据库压力
                        if (totalDownloaded % (1024 * 1024) < bytesRead) {
                            mapper.updateStatus(taskId, 1, totalDownloaded, null);
                        }
                    }
                    fos.flush();
                }

                // 下载完成,标记为成功
                long finalSize = contentLength > 0 ? contentLength : targetPath.toFile().length();
                mapper.markAsSuccess(taskId, targetPath.toString(), finalSize);
                log.info("[Task-{}] 定时下载完成: {} -> {} (总大小: {} 字节)",
                        taskId, task.getImageUrl(), targetPath, finalSize);

            }

        } catch (IOException e) {
            // 断点续传中途异常, 保留已下载进度, 等待下一次定时任务自动重试
            if (task.getDownloadedSize() != null && task.getDownloadedSize() > 0) {
                mapper.updateStatus(taskId, 0, task.getDownloadedSize(), "断点续传中途异常: " + e.getMessage());
                log.warn("[Task-{}] 断点续传中断, 已保存下载进度: {} 字节, 等待下次定时任务重试", taskId, task.getDownloadedSize());
            } else {
                handleFailure(taskId, e.getMessage());
            }
        } catch (Exception e) {
            handleFailure(taskId, e.getMessage());
        }
    }

    /**
     * 下载失败处理,自动执行重试机制
     */
    @Override
    public void handleFailure(long taskId, String errorMsg) {
        log.error("[Task-{}] 图片下载失败: {}", taskId, errorMsg);

        ImageDownloadInfo task = mapper.selectById(taskId);
        if (task == null) return;

        int retryCount = task.getRetryCount();
        if (retryCount < config.getMaxRetry()) {
            // 未达到最大重试次数,重置为待下载状态,等待下一次定时调度自动重试
            mapper.incrementRetryCount(taskId);
            mapper.updateStatus(taskId, 0, task.getDownloadedSize(), errorMsg);
            log.info("[Task-{}] 已自动安排定时重试, 当前重试次数: {}/{}", taskId, retryCount + 1, config.getMaxRetry());
        } else {
            // 达到最大重试次数,标记为永久失败
            mapper.updateStatus(taskId, 3, task.getDownloadedSize(), errorMsg);
            log.error("[Task-{}] 超过最大重试次数{}, 标记为永久失败", taskId, config.getMaxRetry());
        }
    }

    /**
     * 从URL或Content-Disposition提取文件名
     */
    private String extractFileName(String url, Response response, long taskId) {
        // 优先从响应头Content-Disposition获取文件名
        String contentDisp = response.header("Content-Disposition");
        if (contentDisp != null && contentDisp.contains("filename=")) {
            String name = contentDisp.split("filename=")
                    .replace("\"", "").trim();
            if (!name.isEmpty()) {
                return sanitizeFileName(name);
            }
        }

        // 从图片URL中提取文件名
        try {
            String path = new URL(url).getPath();
            String fileName = path.substring(path.lastIndexOf('/') + 1);
            if (fileName.isEmpty() || !fileName.contains(".")) {
                fileName = "image_" + taskId + "_" + System.currentTimeMillis() + ".jpg";
            }
            return sanitizeFileName(fileName);
        } catch (Exception e) {
            return "image_" + taskId + "_" + System.currentTimeMillis() + ".jpg";
        }
    }

    /**
     * 文件名清洗, 防止路径注入和非法字符
     */
    private String sanitizeFileName(String name) {
        return name.replaceAll("[\\\\/:*?\"<>|]", "_")
                .replaceAll("\\s+", "_")
                .substring(0, Math.min(name.length(), 200));
    }

    /**
     * 文件名冲突自动处理, 自动追加序号后缀
     */
    private Path resolveConflict(Path path) {
        if (!Files.exists(path)) return path;
        String name = path.getFileName().toString();
        int dotIndex = name.lastIndexOf('.');
        String base = dotIndex > 0 ? name.substring(0, dotIndex) : name;
        String ext = dotIndex > 0 ? name.substring(dotIndex) : "";

        int counter = 1;
        Path newPath;
        do {
            newPath = path.getParent().resolve(base + "_" + counter++ + ext);
        } while (Files.exists(newPath));

        return newPath;
    }

    @Override
    public void addTask(String imageUrl) {
        mapper.insertTask(imageUrl);
        log.info("成功添加定时图片下载任务: {}", imageUrl);
    }

    @Override
    public Map<String, Object> getStatistics() {
        Map<String, Object> stats = new HashMap<>();
        stats.put("pending", mapper.countPendingTasks());
        stats.put("downloading", mapper.countDownloadingTasks());
        stats.put("maxRetry", config.getMaxRetry());
        stats.put("scanCron", config.getScanCron());
        return stats;
    }
}
```

```java
// ImageDownloadController.java
package com.example.imagedownload.controller;

import com.example.imagedownload.service.ImageDownloadService;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

@RestController
@RequestMapping("/api/images")
@RequiredArgsConstructor
public class ImageDownloadController {

    private final ImageDownloadService downloadService;

    /**
     * 手动添加单个下载任务
     */
    @PostMapping("/download/task")
    public ResponseEntity<String> addTask(@RequestParam String url) {
        downloadService.addTask(url);
        return ResponseEntity.ok("定时下载任务已添加: " + url);
    }

    /**
     * 批量添加下载任务
     */
    @PostMapping("/download/task/batch")
    public ResponseEntity<String> batchAddTask(@RequestBody List<String> urls) {
        urls.forEach(downloadService::addTask);
        return ResponseEntity.ok("批量添加定时下载任务完成, 共 " + urls.size() + " 个任务");
    }

    /**
     * 获取定时下载任务统计信息
     */
    @GetMapping("/download/stats")
    public ResponseEntity<Map<String, Object>> getStats() {
        return ResponseEntity.ok(downloadService.getStatistics());
    }
}
```
