本地下拉式漫画阅读器 - 开发文档

---

1. 项目概述

1.1 项目名称
MangaReader（本地下拉式漫画阅读器）

1.2 项目目标
开发一款 Windows 10 平台的本地漫画阅读器，支持下拉式（垂直滚动）阅读模式，漫画图片存储在本地磁盘，元数据通过 MySQL 管理。

1.3 核心特性
| 特性 | 说明 |
|------|------|
| 下拉式阅读 | 图片纵向拼接，连续滚动阅读 |
| 本地存储 | 漫画图片存放本地磁盘指定目录 |
| MySQL管理 | 元数据（标题、作者、章节、路径等）存入MySQL |
| 漫画管理 | 添加/删除/编辑漫画及章节 |
| 阅读体验 | 双页模式、缩放、书签、阅读进度记忆 |
| 图片格式 | 支持 JPG、PNG、WebP、GIF |

5.1.3 图片加载优化策略

```
图片加载流程：
┌──────────┐    ┌──────────┐    ┌──────────┐    ┌──────────┐
│ 检查内存 │───▶│ 命中？   │───▶│ 磁盘缓存 │───▶│ 磁盘读取 │
│ LRU缓存  │    │ 返回     │    │ 命中？   │    │ 解码缩放 │
└──────────┘    └──────────┘    └──────────┘    └──────────┘
                     │                               │
                     ▼                               ▼
               直接返回                        写入磁盘缓存
                                             并加入内存缓存
```

```java
public class ImageService {
    // 内存LRU缓存（最多缓存50张图片）
    private LRUCache<String, Image> memoryCache;
    
    // 磁盘缓存目录
    private String diskCacheDir = "cache/images/";
    
    public Image loadImage(String path, double scale) {
        // 1. 查内存缓存
        Image img = memoryCache.get(path);
        if (img != null) return img;
        
        // 2. 查磁盘缓存
        String cacheKey = path + "_" + scale;
        File cacheFile = new File(diskCacheDir + cacheKey + ".dat");
        if (cacheFile.exists()) {
            img = loadFromDiskCache(cacheFile);
            memoryCache.put(path, img);
            return img;
        }
        
        // 3. 从磁盘读取并解码
        img = decodeAndScale(path, scale);
        
        // 4. 写入磁盘缓存
        saveToDiskCache(img, cacheFile);
        
        // 5. 加入内存缓存
        memoryCache.put(path, img);
        return img;
    }
}
```

7. 核心类设计

7.1 自定义 LongImageView（下拉核心）

```java
public class LongImageView extends Pane {
    
    private final ObservableList<Image> images = FXCollections.observableArrayList();
    private final ImageService imageService;
    private double scale = 1.0;
    private boolean dualPageMode = false;
    
    // 可见范围的图片索引
    private int firstVisible = 0;
    private int lastVisible = 0;
    
    public LongImageView(ImageService imageService) {
        this.imageService = imageService;
        
        // 监听滚动
        getParent().sceneProperty().addListener((obs, old, scene) -> {
            if (scene != null) {
                scene.widthProperty().addListener((w, o, n) -> {
                    recalculateLayout();
                });
            }
        });
    }
    
    /**
     * 设置章节图片路径列表
     */
    public void setImagePaths(List<String> imagePaths, double scale) {
        this.scale = scale;
        images.setAll(
            imagePaths.stream()
                .map(p -> imageService.loadImage(p, scale))
                .collect(Collectors.toList())
        );
        recalculateLayout();
    }
    
    /**
     * 计算布局：只布局可见区域内的图片
     */
    @Override
    protected void layoutChildren() {
        super.layoutChildren();
        double y = 0;
        double maxWidth = getWidth();
        
        // 获取父ScrollPane的可视区域
        ScrollPane sp = (ScrollPane) getParent();
        double viewHeight = sp.getViewportBounds().getHeight();
        double viewportY = sp.getVvalue() * (getHeight() - viewHeight);
        
        for (int i = 0; i < images.size(); i++) {
            Image img = images.get(i);
            if (img == null || img.getWidth() <= 0) continue;
            
            double w = img.getWidth() * scale;
            double h = img.getHeight() * scale;
            
            // 跳过不可见的
            if (y + h < viewportY) {
                y += h;
                continue;
            }
            if (y > viewportY + viewHeight + h) break;
            
            // 布局
            double x = (maxWidth - w) / 2;  // 居中
            layoutInArea(createImageNode(img, w, h), x, y, w, h, 0, HPos.CENTER, VPos.TOP);
            y += h;
        }
        
        // 清理不需要的子节点
        removeUnnecessaryChildren(viewportY, viewHeight);
    }
    
    private void recalculateLayout() {
        requestLayout();
    }
    
    /**
     * 获取当前阅读位置（图片索引 + 像素偏移）
     */
    public ReadingPosition getReadingPosition() {
        // 计算当前视口中心对应的图片和偏移
        ScrollPane sp = (ScrollPane) getParent();
```

```java
public class ReaderController {
    
    @FXML private LongImageView longImageView;
    @FXML private ScrollPane scrollPane;
    @FXML private Label pageLabel;
    @FXML private Slider progressSlider;
    
    private Chapter currentChapter;
    private Manga currentManga;
    private final ProgressService progressService;
    
    @FXML
    public void initialize() {
        // 滚动监听 -> 保存进度
        scrollPane.vvalueProperty().addListener((obs, old, newVal) -> {
            ReadingPosition pos = longImageView.getReadingPosition();
            progressService.saveProgress(
                currentManga.getId(), 
                currentChapter.getId(), 
                pos.getPageIndex(), 
                pos.getOffset()
            );
        });
        
        // 键盘快捷键
        scrollPane.setOnKeyPressed(e -> {
            switch (e.getCode()) {
                case UP: case PAGE_UP: prevPage(); break;
                case DOWN: case PAGE_DOWN: nextPage(); break;
                case PLUS: case EQUALS: zoomIn(); break;
                case MINUS: zoomOut(); break;
                case D: toggleDualPage(); break;
            }
        });
    }
    
    public void openChapter(Manga manga, Chapter chapter) {
        this.currentManga = manga;
        this.currentChapter = chapter;
        
        // 加载图片
        List<String> paths = chapterService.getImagePaths(chapter);
        longImageView.setImagePaths(paths, currentScale);
        
        // 恢复进度
        ReadingProgress progress = progressService.getProgress(manga.getId(), chapter.getId());
        if (progress != null) {
            scrollToPosition(progress.getPageIndex(), progress.getReadPosition());
        }
        
        pageLabel.setText("第 " + chapter.getChapterNum() + " 章");
        progressSlider.setMax(paths.size());
    }
    
    private void nextPage() {
        // 向下滚动一页
        double v = scrollPane.getVvalue();
        scrollPane.setVvalue(Math.min(v + 0.1, 1.0));
    }
    
    private void prevPage() {
        double v = scrollPane.getVvalue();
        scrollPane.setVvalue(Math.max(v - 0.1, 0.0));
    }
}
```

8. 性能优化方案

| 问题 | 方案 |
|------|------|
| 大章节内存溢出 | 虚拟滚动 + 按需加载 + 离屏回收 |
| 图片解码慢 | 后台线程异步解码 + 磁盘缓存 |
| 滚动卡顿 | 双缓冲渲染 + 图片预加载（预读前后20张） |
| 启动慢 | 懒加载元数据，首次扫描后缓存到本地 |
| 频繁数据库访问 | 连接池 + 批量查询 + 本地进度缓存 |

```java
// 预加载策略：始终保持视口上下各 N 张已加载
private void ensurePreload(int visibleStart, int visibleEnd) {
    int preloadBefore = Math.max(0, visibleStart - PRELOAD_COUNT);
    int preloadAfter = Math.min(images.size() - 1, visibleEnd + PRELOAD_COUNT);
    
    for (int i = preloadBefore; i <= preloadAfter; i++) {
        if (images.get(i) == null || images.get(i).getWidth() <= 0) {
            // 异步加载
            imageService.loadImageAsync(paths.get(i), scale, img -> {
                Platform.runLater(() -> {
                    images.set(i, img);
                });
            });
        }
    }
}
```
